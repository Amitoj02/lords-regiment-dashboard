# Lords Regiment — https://lordsofholdfast.com
#
# This file is static and ships with the Angular build: public/ is copied
# verbatim to the dist root by angular.json and served by nginx like any other
# asset, so a normal deploy publishes it and no edge config has to change.
#
# The sitemap below is NOT a static file next to this one, and cannot be: it
# enumerates every public profile from live data, so the API generates it at
# /api/sitemap.xml and Caddy carries an explicit `handle /sitemap.xml` block
# that proxies the apex path to it. Adding a static sitemap.xml here would
# shadow that route with a file that is stale the moment a member joins.

User-agent: *
Allow: /

# The staff dashboard and every auth hop. None of these have anything to index,
# and a crawler following them only burns budget on redirects to Discord.
Disallow: /app/
Disallow: /login
Disallow: /auth/
Disallow: /onboarding/
Disallow: /guild-required

# A member's own settings. Reachable only with their session, so a crawler
# would never see anything but the sign-in redirect.
Disallow: /account

# "My profile" — authGuard'd, and it resolves to the caller's own /u/@handle,
# which IS indexable. Without this the two URLs compete for the same page and
# a crawler spends its budget on the one that only ever redirects.
Disallow: /me

# ── Link unfurlers (T-0293) ─────────────────────────────────────────────────
#
# Whether Discordbot honours robots.txt is genuinely contested — its own issue
# tracker carries reports of a too-restrictive robots.txt suppressing embeds,
# and third-party UA registries claim the opposite. The `User-agent: *` block
# above already allows everything these need, so this group changes nothing
# today; it exists so that a future `Disallow:` added for human-facing reasons
# cannot silently take every link preview on the site down with it.
#
# Slack is here for completeness, not necessity: it documents that it ignores
# robots.txt outright, on the grounds that its fetch is made on behalf of a
# person who just pasted the link.

User-agent: Discordbot
Allow: /

User-agent: Twitterbot
Allow: /

User-agent: Slackbot-LinkExpanding
Allow: /

User-agent: facebookexternalhit
Allow: /
